== 2. XFS の末尾の空きを縮める試験的な機能（第9章の注の出どころ） ==
@中
$ sudo pvcreate -q /dev/vdc
$ sudo vgcreate -q vgx /dev/vdc
$ sudo lvcreate -q -L 300M -n xfs vgx
$ sudo mkfs.xfs -q /dev/vgx/xfs
$ sudo mkdir /srv/x
$ sudo mount /dev/vgx/xfs /srv/x
$ sudo xfs_info /srv/x | grep "^data"
$ sudo xfs_growfs -D 70000 /srv/x
$ sudo xfs_info /srv/x | grep "^data"
$ sudo umount /srv/x
