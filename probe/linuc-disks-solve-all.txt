#   2) 模範解答どおりにやれば ✔ になること
#   3) 課題文の「いまの状態」に載せる画面を、読者と同じ打ち方で取ること
# 練習機は工場出荷時の状態から始め、演習だけを 01 から順にやる人の道筋をたどる（理論編の章は打たない）。

$ sudo fdisk /dev/vde
@keys n|p|1|||w

$ lab check 01
$ sudo mkfs.ext4 -q /dev/vde1

$ lab check 01
$ sudo umount /mnt
$ lab check 01

== 4. 演習04 — 5種類のファイルシステムを作り、ext4 の予約を減らす ==

== 5. 演習05 — ディスクのイメージのファイルから、中のファイルを取り出す ==
@中
$ ls -l ~/old-usb.img
$ lab check 05
$ sudo blkid ~/old-usb.img
$ sudo mount -o loop ~/old-usb.img /mnt
$ losetup -l
$ ls -l /mnt
# --- マウントしただけ（まだ取り出していない）---
$ lab check 05
$ cp /mnt/message.txt ~/
# --- 取り出したが、まだマウントしたまま ---
$ lab check 05
$ sudo umount /mnt
$ cat ~/message.txt
$ lab check 05
# 模範解答の「-o loop を付けずに mount したら」の確かめ
$ sudo mount ~/old-usb.img /mnt
$ losetup -l
$ sudo umount /mnt

== 7. 演習07 — スワップを作り、起動しても有効にする ==
@中

$ sudo mkswap -L SWAP /dev/vdb3
$ sudo swapon /dev/vdb3
$ swapon --show
# --- swapon しただけ（/etc/fstab に無い）---
$ lab check 07
# --- 判定の後も、vdb3 は判定の前と同じく使われているか ---
$ swapon --show
# --- /etc/fstab にデバイス名で書いた ---

$ lab check 07
$ sudo sed -i '/^\/dev\/vdb3 /d' /etc/fstab

$ sudo systemctl daemon-reload
$ lab check 07

== 8. 演習08 — 2台のディスクにまたがる箱を作り、中身ごと広げる ==
@中
$ lsblk /dev/vdc /dev/sda
$ lab check 08
$ sudo pvcreate /dev/vdc /dev/sda
$ sudo vgcreate vg0 /dev/vdc /dev/sda
$ sudo lvcreate -L 600M -n data vg0
$ sudo mkfs.ext4 -q /dev/vg0/data
$ sudo mkdir /srv/data
$ sudo mount /dev/vg0/data /srv/data
# --- 1台に収まる大きさのまま ---
$ lab check 08

$ df -h /srv/data
# --- 箱（LV）だけ広げた ---
$ lab check 08
$ sudo resize2fs /dev/vg0/data
$ df -h /srv/data
$ lab check 08

== 9. 演習09 — vdb を SSD として扱わせる（再起動しても） ==
@中
$ lsblk -d -o NAME,TRAN,ROTA,SIZE,MODEL
$ lab check 09
$ echo 0 | sudo tee /sys/block/vdb/queue/rotational
# --- 手で 0 を書いただけ（再起動すると戻る状態）---
$ lab check 09
# --- 判定の後、値は判定の前の 0 に戻っているか ---
$ cat /sys/block/vdb/queue/rotational
$ echo 'ACTION=="add|change", SUBSYSTEM=="block", KERNEL=="vdb", ATTR{queue/rotational}="0"' | sudo tee /etc/udev/rules.d/60-lab-ssd.rules
$ sudo udevadm trigger --action=add --subsystem-match=block --sysname-match=vdb --settle
$ cat /sys/block/vdb/queue/rotational
$ lab check 09

== 10. 演習10 — USB メモリを挿し直しても使えなくする ==
@外
$ virsh attach-device linuc ~/vm/extra/usb-a.xml
@中
$ lsblk -d -o NAME,TRAN,SIZE,SERIAL
$ lsmod | grep -E "usb_storage|uas"
@外
$ virsh detach-device linuc ~/vm/extra/usb-a.xml
@中
$ lab check 10
$ echo "blacklist usb_storage" | sudo tee /etc/modprobe.d/lab-usb.conf
$ sudo modprobe -r uas
# --- blacklist usb_storage だけ（uas の依存で読み込まれるはず）---
$ lab check 10
$ printf "blacklist usb_storage\nblacklist uas\n" | sudo tee /etc/modprobe.d/lab-usb.conf
$ sudo modprobe -r uas
# --- 同じファイルを書き換えて uas も足しただけ（udev が書き換えに気づかないはず）---
$ lab check 10
$ sudo systemctl restart systemd-udevd
$ sudo modprobe -r uas
# --- ヒント3のとおり udev を再起動した ---
$ lab check 10
# --- ファイルを消しただけ（udev が気づいて、また使えるようになるはず）---
$ sudo rm /etc/modprobe.d/lab-usb.conf
$ lab check 10
# --- 模範解答: 新しいファイルに2行（udev の再起動なし）---
$ printf "blacklist usb_storage\nblacklist uas\n" | sudo tee /etc/modprobe.d/lab-usb.conf
$ sudo modprobe -r uas
$ lab check 10
# --- 模範解答の確かめ方: 外で挿して、中で見て、外で抜く ---
@外
$ virsh attach-device linuc ~/vm/extra/usb-a.xml
@中
$ lsusb -t
$ lsmod | grep -E "usb_storage|uas"
@外
$ virsh detach-device linuc ~/vm/extra/usb-a.xml
@中
