== 4. 演習04 の境目 ==
@中
# --- vdb1 を -F 32 なしで作った（100MiB なので FAT16 になる）。ほかは課題どおり ---
$ sudo mkfs.vfat /dev/vdb1
$ sudo mkfs -t ext3 -L NEWBOOT /dev/vdb2
$ sudo mkfs.xfs -L ROOT /dev/vdb4
$ sudo mkfs.btrfs -L VAR /dev/vdb5
$ sudo mkfs.ext4 -L HOME /dev/vdb6
$ sudo tune2fs -m 1 /dev/vdb6
$ sudo blkid -p -o export /dev/vdb1
$ lab check 04
# --- vdb1 を -F 32 で作り直した（ほかは課題どおり）---
$ sudo mkfs.vfat -F 32 /dev/vdb1
$ lab check 04
# --- vdb2 のラベルを BOOT にした ---
$ sudo e2label /dev/vdb2 BOOT
$ lab check 04
# --- vdb2 のラベルを BOOT 以外（MYBOOT）にした ---
$ sudo e2label /dev/vdb2 MYBOOT
$ lab check 04
# --- vdb2 のラベルを消した ---
$ sudo e2label /dev/vdb2 ""
$ lab check 04
# --- vdb2 を ext3 でなく FAT32（ラベル MYBOOT）にした（課題の種類と違う）---
$ sudo mkfs.vfat -F 32 -n MYBOOT /dev/vdb2
$ lab check 04
# --- 戻す（先に印を消してから ext3）---
$ sudo wipefs -a /dev/vdb2
$ sudo mkfs -t ext3 -q -L NEWBOOT /dev/vdb2
$ lab check 04

== 7. 演習07 の境目 ==
@中
$ sudo mkswap -L SWAP /dev/vdb3
# --- /etc/fstab に UUID で書いたが、swapon していない ---
$ echo "UUID=$(sudo blkid -s UUID -o value /dev/vdb3)  none  swap  sw  0  0" | sudo tee -a /etc/fstab
$ sudo systemctl daemon-reload
$ swapon --show
$ lab check 07
# --- 判定のあと、vdb3 は判定の前と同じく使われていないか ---
$ swapon --show
# --- 優先度 5 で使い始めた ---
$ sudo swapon -p 5 /dev/vdb3
$ swapon --show
$ lab check 07
$ swapon --show
# --- /etc/fstab から行を消した（使ってはいる）---
$ sudo sed -i '/ swap /d' /etc/fstab
$ sudo systemctl daemon-reload
$ lab check 07
# --- 判定のあと、vdb3 はまた使われているか（優先度は 5 のままか）---
$ swapon --show

== 9. 演習09 の境目 ==
@中

$ cat /sys/block/vdb/queue/rotational
$ lab check 09
$ cat /sys/block/vdb/queue/rotational
# --- 規則を change だけにした ---
$ echo 'ACTION=="change", SUBSYSTEM=="block", KERNEL=="vdb", ATTR{queue/rotational}="0"' | sudo tee /etc/udev/rules.d/60-lab-ssd.rules

$ cat /sys/block/vdb/queue/rotational
$ lab check 09
$ cat /sys/block/vdb/queue/rotational
