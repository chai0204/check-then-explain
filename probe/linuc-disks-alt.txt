== 3. 演習03（別解: parted で切る。先に esp の旗を付け忘れる） ==
@中
$ sudo parted -s /dev/vdb mklabel gpt
$ sudo parted -s /dev/vdb mkpart ESP fat32 1MiB 101MiB
$ sudo parted -s /dev/vdb mkpart boot ext4 101MiB 401MiB
$ sudo parted -s /dev/vdb mkpart swap linux-swap 401MiB 657MiB
$ sudo parted -s /dev/vdb mkpart root ext4 657MiB 1257MiB
$ sudo parted -s /dev/vdb mkpart var ext4 1257MiB 1657MiB
$ sudo parted -s /dev/vdb mkpart home ext4 1657MiB 100%
@sleep 3
# --- parted で6つ作ったが、1番に esp の旗を付けていない ---
$ lab check 03
$ sudo parted -s /dev/vdb set 1 esp on
@sleep 3
$ sudo parted /dev/vdb print
# --- 別解: parted で6つ・1番に esp ---
$ lab check 03

$ lab check 04

== 8. 演習08（別解: 区画（種類 8e）を物理ボリュームにする） ==

== 9. 演習09（別解1: tmpfiles.d で起動のたびに書く → 再起動して確かめる。別解2: 製造番号で当てる udev の規則） ==
@中
$ echo "w /sys/block/vdb/queue/rotational - - - - 0" | sudo tee /etc/tmpfiles.d/fc-ssd.conf
$ sudo reboot
@外
$ vm start
$ ssh linuc
@中
$ cat /sys/block/vdb/queue/rotational
$ lsblk -d -o NAME,ROTA /dev/vdb
# --- 別解1: 再起動したら SSD のまま（tmpfiles.d）---
$ lab check 09
$ sudo rm /etc/tmpfiles.d/fc-ssd.conf
$ echo 'ACTION=="add|change", SUBSYSTEM=="block", ENV{DEVTYPE}=="disk", ATTRS{serial}=="LAB-VDB", ATTR{queue/rotational}="0"' | sudo tee /etc/udev/rules.d/61-fc-ssd.rules
$ sudo udevadm trigger --action=add --subsystem-match=block --sysname-match=vdb --settle
$ cat /sys/block/vdb/queue/rotational
# --- 別解2: 製造番号（LAB-VDB）で当てる udev の規則 ---
$ lab check 09
