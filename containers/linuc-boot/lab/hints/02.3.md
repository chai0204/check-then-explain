    ssh-keygen -t rsa                       # Enter を3回
    ssh-copy-id -i ~/.ssh/id_rsa.pub linuc  # 初回は yes、パスワードは student
    ssh linuc                               # パスワードを聞かれなければ成功
    exit

それでもパスワードを聞かれるときは、練習機の側で権限を見てください。
~/.ssh や authorized_keys が他人（other）に書ける権限だと、sshd はその鍵を使わず、パスワードに戻ります
（本家の OpenSSH は group に書けるだけでも断ります。Ubuntu は「グループが本人だけなら許す」改造をしています）。
理由は練習機のログに出ます。

    ssh linuc 'ls -ld ~/.ssh; ls -l ~/.ssh/authorized_keys'
    ssh linuc "sudo journalctl -u ssh -n 20 --no-pager | grep -i refused"
