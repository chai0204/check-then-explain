    #!/bin/bash
    echo "1) ユーザ名  2) ホスト名  q) 終了"
    while read -r ans; do
        case "$ans" in
            1) id -un ;;
            2) hostname ;;
            q) echo "終了します"; exit 0 ;;
            *) echo "1・2・q のどれかを入力してください" ;;
        esac
    done
